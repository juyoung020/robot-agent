# sgclip: SigLIP 2 B/32-256 물체 영상 임베딩(TensorRT FP16 + CUDA 자르기 커널) + 라벨 표 찾기(CPU SIMD) + 기억 폴더 캐시.
#   cmake -S src/scene_graph/clip -B ~/sgclip_build && cmake --build ~/sgclip_build -j && ctest --test-dir ~/sgclip_build
# runtime(libsgrt) 가 add_subdirectory 로 같이 빌드한다. TensorRT 8.2(Jetson Nano, JetPack 4.6) 와 10(PC) 둘 다.
cmake_minimum_required(VERSION 3.18)
if(NOT DEFINED OVDET_CUDA_ROOT)
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64")
    set(OVDET_CUDA_ROOT "/usr/local/cuda" CACHE PATH "CUDA toolkit root")
  else()
    set(OVDET_CUDA_ROOT "/usr/local/cuda-12.8" CACHE PATH "CUDA toolkit root")
  endif()
endif()
if(NOT CMAKE_CUDA_COMPILER)
  set(CMAKE_CUDA_COMPILER ${OVDET_CUDA_ROOT}/bin/nvcc)
endif()
if(NOT CMAKE_CUDA_ARCHITECTURES)
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64")
    set(CMAKE_CUDA_ARCHITECTURES 53 CACHE STRING "GPU architectures")   # Jetson Nano(Maxwell)
  else()
    set(CMAKE_CUDA_ARCHITECTURES 120 CACHE STRING "GPU architectures")
  endif()
endif()
project(sgclip LANGUAGES CXX CUDA)
set(CUDAToolkit_ROOT ${OVDET_CUDA_ROOT})
find_package(CUDAToolkit REQUIRED)
if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
endif()
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

find_path(SGC_TRT_INCLUDE NvInfer.h HINTS /usr/include/x86_64-linux-gnu /usr/include/aarch64-linux-gnu REQUIRED)
find_library(SGC_NVINFER nvinfer HINTS /usr/lib/x86_64-linux-gnu /usr/lib/aarch64-linux-gnu REQUIRED)
find_path(SGC_JSON_INCLUDE nlohmann/json.hpp REQUIRED)
find_package(PNG)   # 물체 찾기 대체 벡터: 기억의 best view 사진(objects/O<id>_rgb.png·_mask.png). 없으면 대체 벡터 뽑기만 꺼짐   # 헤더 하나(JetPack 4.6: apt nlohmann-json3-dev 또는 같이 둠)

add_library(sgclip STATIC src/crop.cu src/encoder.cpp src/labels.cpp src/memstore.cpp src/textenc.cpp src/objindex.cpp)
set_target_properties(sgclip PROPERTIES POSITION_INDEPENDENT_CODE ON CUDA_RUNTIME_LIBRARY Static)
target_include_directories(sgclip PUBLIC include PRIVATE src)
target_include_directories(sgclip SYSTEM PRIVATE ${SGC_TRT_INCLUDE} ${SGC_JSON_INCLUDE})
target_link_libraries(sgclip PUBLIC ${SGC_NVINFER} CUDA::cudart_static ${CMAKE_DL_LIBS} pthread)
if(PNG_FOUND)
  target_link_libraries(sgclip PUBLIC PNG::PNG)
  target_compile_definitions(sgclip PRIVATE SGS_HAVE_PNG=1)
endif()
target_compile_options(sgclip PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-O3 -Wall -Wextra> $<$<COMPILE_LANGUAGE:CUDA>:-O3 -lineinfo>)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
  # 라벨 찾기 SIMD: AVX2 + FMA + F16C(일반 판은 -DSGC_SCALAR=ON)
  option(SGC_SCALAR "label lookup without SIMD" OFF)
  if(NOT SGC_SCALAR)
    set_source_files_properties(src/labels.cpp src/objindex.cpp PROPERTIES COMPILE_OPTIONS "-mavx2;-mfma;-mf16c;-mpopcnt")
  endif()
endif()
