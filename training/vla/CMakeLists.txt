cmake_minimum_required(VERSION 3.22)
# RecallVLA (docs/map_vla/MAPVLA_SPEC.md): Qwen3.5 몸통 + SigLIP 2 + 지도 인코더 + 행동 전문가. RL network 의 손 GEMM(gemm.cuh)을 include 로만 쓴다.
if(NOT CMAKE_CUDA_COMPILER AND EXISTS /usr/local/cuda-12.8/bin/nvcc)
  set(CMAKE_CUDA_COMPILER /usr/local/cuda-12.8/bin/nvcc)
endif()
project(recallvla LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CUDA_ARCHITECTURES 120)
set(CMAKE_CXX_FLAGS_RELEASE "-O2 -ffp-contract=off")
set(CMAKE_CUDA_FLAGS_RELEASE "-O3 --fmad=false -prec-div=true -prec-sqrt=true -ftz=false")
find_package(CUDAToolkit REQUIRED)
find_package(OpenMP REQUIRED)
set(RLNET ${CMAKE_CURRENT_LIST_DIR}/../RL/network)
set(INC ${CMAKE_CURRENT_LIST_DIR}/include ${CMAKE_CURRENT_LIST_DIR}/src ${RLNET}/include ${RLNET}/src)

add_library(rvla STATIC src/st.cpp src/qwen.cu src/qkern.cu src/tkern.cu src/vgemm.cu src/dnchunk.cu src/flash.cu src/model.cu src/optim.cu)
target_include_directories(rvla PUBLIC ${INC})
target_link_libraries(rvla PUBLIC CUDA::cudart)

add_executable(qwen_verify tools/qwen_verify.cu)
target_link_libraries(qwen_verify rvla)

# 검증 전용 CPU 참조판(V5)과 학습 검증·측정 도구
add_library(vref STATIC tools/vref.cpp)
target_include_directories(vref PUBLIC ${INC} tools ${CUDAToolkit_INCLUDE_DIRS})
target_compile_options(vref PRIVATE -O3)
add_executable(vla_verify tools/vla_verify.cu)
target_include_directories(vla_verify PRIVATE tools)
target_link_libraries(vla_verify rvla vref)

enable_testing()
add_test(NAME vla_v5 COMMAND vla_verify v5)
add_test(NAME vla_v5_ki0 COMMAND vla_verify v5 --ki0)
add_test(NAME vla_v5_frozen_vis COMMAND vla_verify v5 --frozen-vis)
add_test(NAME vla_negative COMMAND vla_verify neg)
add_test(NAME vla_v67 COMMAND vla_verify v67)
add_test(NAME vla_opt COMMAND vla_verify opt)
add_test(NAME vla_dnref COMMAND vla_verify dnref)
add_test(NAME vla_dn COMMAND vla_verify dn)
add_test(NAME vla_att COMMAND vla_verify att)
add_test(NAME vla_v5_mem COMMAND vla_verify v5 --mem)
add_test(NAME vla_v5_mem_ki0 COMMAND vla_verify v5 --mem --ki0)
add_test(NAME vla_negative_mem COMMAND vla_verify neg --mem)
add_test(NAME vla_v67_mem COMMAND vla_verify v67 --mem)
