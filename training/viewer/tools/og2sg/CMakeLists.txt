cmake_minimum_required(VERSION 3.18)
# og2sg — OmniGibson LIMO 탐사 기록(rec.bin) → 학습 뷰어 "sgview 판"(.sg). scenemap(C++, CPU)은 src/scene_graph 소스를 그대로 빌드해 C ABI 로만 쓴다(읽기만).
#   tools/build_all.sh og2sg  (→ build/og2sg)
project(og2sg LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
# 실제 파이프라인 libsgrt(tools/build_all.sh sgrt → build/bin/libsgrt.so) 하나에 링크 — scenemap(sm_*)도 그 안의 것을 쓴다
set(RA_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../..")
find_library(SGRT_LIB sgrt PATHS ${RA_ROOT}/build/bin NO_DEFAULT_PATH REQUIRED)
add_executable(og2sg og2sg.cpp)
target_compile_definitions(og2sg PRIVATE RA_ROOT_DIR="${RA_ROOT}")
target_include_directories(og2sg PRIVATE ${RA_ROOT}/src/scene_graph/common ${RA_ROOT}/src/scene_graph/runtime/include ${RA_ROOT}/src/scene_graph/scenemap/include ${CMAKE_CURRENT_LIST_DIR}/../record_replay ${CMAKE_CURRENT_LIST_DIR}/../../../RL/tools/b1kconv/cpp)
target_link_libraries(og2sg ${SGRT_LIB} pthread)
set_target_properties(og2sg PROPERTIES BUILD_RPATH "${RA_ROOT}/build/bin" INSTALL_RPATH "${RA_ROOT}/build/bin")
