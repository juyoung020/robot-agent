cmake_minimum_required(VERSION 3.22)
# pick_cmp: GPU 지도 판 내보내기·목표 원인(README). 진짜 쪽은 training/viewer/tools/og_replay/og_cmp.py.
if(NOT CMAKE_CUDA_COMPILER AND EXISTS /usr/local/cuda-12.8/bin/nvcc)
  set(CMAKE_CUDA_COMPILER /usr/local/cuda-12.8/bin/nvcc)
endif()
project(map_cmp LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CUDA_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CUDA_ARCHITECTURES 120)
set(CMAKE_CXX_FLAGS_RELEASE "-O2 -ffp-contract=off")
set(CMAKE_CUDA_FLAGS_RELEASE "-O3 --fmad=false -prec-div=true -prec-sqrt=true -ftz=false")
find_package(CUDAToolkit REQUIRED)
find_package(OpenMP REQUIRED)

set(RL_DIR ${CMAKE_CURRENT_LIST_DIR}/.. CACHE PATH "training/RL (env·map 소스, 기본 이 저장소)")
add_library(cmp_env STATIC ${RL_DIR}/env/src/env_kernel.cu ${RL_DIR}/env/src/env_ref.cpp)
target_include_directories(cmp_env PUBLIC ${RL_DIR}/env/include)

# 집을 물체 확정 진단 + 진짜 파이프라인 비교용 판 내보내기(README "pick_cmp"). 학습과 같은 근사판(잡음 켬, 소스 그대로)
add_library(gmap_train STATIC ${RL_DIR}/map/src/map_kernel.cu ${RL_DIR}/map/src/map_ref.cpp)
target_include_directories(gmap_train PUBLIC ${RL_DIR}/map/include)
target_link_libraries(gmap_train PUBLIC cmp_env CUDA::cudart OpenMP::OpenMP_CXX)
add_library(cmp_bscene STATIC ${RL_DIR}/env/src/bscene_host.cpp)
target_include_directories(cmp_bscene PUBLIC ${RL_DIR}/env/include ${RL_DIR}/tools/b1kconv/cpp)
target_link_libraries(cmp_bscene PUBLIC CUDA::cudart)
add_executable(pick_cmp tools/pick_cmp.cu)
target_include_directories(pick_cmp PRIVATE ${RL_DIR}/../viewer/tools/record_replay)
target_link_libraries(pick_cmp gmap_train cmp_bscene)
